// ✅ validationService.js - PLZ Validierung
static validateZip(value) {
  if (!value || value.trim().length === 0) {
    return { isValid: false, error: 'Postleitzahl erforderlich' }
  }
  if (!CONFIG.PATTERNS.ZIP_CODE.test(value)) {
    return { isValid: false, error: 'Muss 5 Ziffern sein' }
  }
  // Bereichsprüfung: Erste 2 Ziffern müssen 69 sein
  const prefix = value.substring(0, 2)
  const expected = CONFIG.BUSINESS_LOCATION.plzPrefix
  if (prefix !== expected) {
    return { isValid: false, error: 'Außerhalb Abholgebiet' }
  }
  return { isValid: true, error: null }
}
✅ Tiefere Validierung: Format + Bereichsprüfung (69xxx)
